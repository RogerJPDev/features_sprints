Operators & control flow

Comparison


                1 === 1        // true
                1 === "1"      // false (different types)
                1 == "1"       // true  (type coercion, avoid relying on this)
                null ?? "default"   // "default" — nullish coalescing
                user?.address?.city // optional chaining, safe on undefined
                user?.getName?.()   // optional call, safe if method doesn't exist

Logical operators & short-circuiting


                true && "yes"    // "yes" (returns second value if first is truthy)
                false && "yes"   // false (short-circuits, "yes" never evaluated)
                null || "default" // "default"
                0 || "fallback"    // "fallback" — 0 is falsy, use ?? if 0 is valid
                const isAdmin = user && user.role === "admin";

call, apply & bind


                function greet() { return `Hi, ${this.name}`; }
                const user = { name: "Ada" };

                greet.call(user);          // "Hi, Ada" — args passed individually
                greet.apply(user, []);     // same, but args passed as an array
                const bound = greet.bind(user);
                bound();                    // "Hi, Ada" — returns a new function, permanently bound

Closures


                function makeCounter() {
                  let count = 0;
                  return function () {
                    count++;
                    return count;
                  };
                }

                const counter = makeCounter();
                counter(); // 1
                counter(); // 2

Currying & partial application


                const multiply = (a) => (b) => a * b;
                const double = multiply(2);
                double(5); // 10

                function curry(fn) {
                  return (...args) =>
                    args.length >= fn.length
                      ? fn(...args)
                      : (...more) => curry(fn)(...args, ...more);
                }

Arrays & objects

Array methods: transforming


                const nums = [1, 2, 3, 4, 5];

                nums.map(n => n * 2);          // [2,4,6,8,10]
                nums.filter(n => n % 2 === 0); // [2,4]
                nums.reduce((sum, n) => sum + n, 0); // 15
                nums.flatMap(n => [n, n * 10]); // [1,10,2,20,...]
                [[1,2],[3,[4]]].flat(2);          // [1,2,3,4]

Array methods: finding & checking


                nums.find(n => n > 3);          // 4
                nums.findIndex(n => n > 3);      // 3
                nums.some(n => n > 4);            // true
                nums.every(n => n > 0);            // true
                nums.includes(3);                    // true
                [...nums].sort((a, b) => a - b);       // copy, then sort ascending
                [...nums].reverse();

Asynchronous JavaScript

Promises


                const promise = new Promise((resolve, reject) => {
                  setTimeout(() => resolve("done"), 1000);
                });

                promise
                  .then(result => console.log(result))
                  .catch(error => console.error(error))

async / await


                async function getUser(id) {
                  try {
                    const response = await fetch(`/api/users/${id}`);
                    if (!response.ok) throw new Error('Request failed');
                    const data = await response.json();
                    return data;
                  } catch (error) {
                    console.error(error);
                  }
                }

fetch with options


                const response = await fetch('/api/users', {
                  method: 'POST',
                  headers: { 'Content-Type': 'application/json' },
                  body: JSON.stringify({ name: 'Ada' }),
                });
                const data = await response.json();

                const [users, posts] = await Promise.all([

Debounce & throttle


                function debounce(fn, delay) {
                  let timer;
                  return (...args) => {
                    clearTimeout(timer);
                    timer = setTimeout(() => fn(...args), delay);
                  };
                }

                input.addEventListener('input', debounce(handleSearch, 300));

Memoization


                function memoize(fn) {
                  const cache = new Map();
                  return (...args) => {
                    const key = JSON.stringify(args);
                    if (!cache.has(key)) cache.set(key, fn(...args));
                    return cache.get(key);
                  };
                }

Generators & async iteration


                function* idGenerator() {
                  let id = 1;
                  while (true) yield id++;
                }
                const gen = idGenerator();
                gen.next().value; // 1
                gen.next().value; // 2

                async function* fetchPages(url) {
                  let next = url;
                  while (next) {
                    const res = await fetch(next).then(r => r.json());
                    yield res.items;
                    next = res.nextPage;
                  }
                }
